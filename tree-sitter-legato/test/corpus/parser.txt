================================================================================
patch with virtual ports and templates
================================================================================

patch voice(freq = 440.0) {
    in gate freq_in

    audio {
        sine: osc { freq: $freq },
        adsr: env { attack: 100.0 }
    }

    freq_in >> osc.freq
    gate >> env.gate
    osc >> env[1]

    { env }
}

patches {
    voice: v1 { freq: 880.0 }
}

{ v1 }
